sealed interface Expr
class Num(val value: int) : Expr
class Add(val left: Expr, val right: Expr) : Expr
class Mul(val left: Expr, val right: Expr) : Expr
class Neg(val operand: Expr) : Expr
class Pick(val cond: Expr, val yes: Expr, val no: Expr) : Expr
fun evaluate(e: Expr): int = when (e) {
    is Num -> e.value
    is Add -> evaluate(e.left) + evaluate(e.right)
    is Mul -> evaluate(e.left) * evaluate(e.right)
    is Neg -> -evaluate(e.operand)
    is Pick -> if (evaluate(e.cond) > 0) evaluate(e.yes) else evaluate(e.no)
}
fun render(e: Expr): string = when (e) {
    is Num -> "${e.value}"
    is Add -> "(${render(e.left)}+${render(e.right)})"
    is Mul -> "(${render(e.left)}*${render(e.right)})"
    is Neg -> "-${render(e.operand)}"
    is Pick -> "(${render(e.cond)}?${render(e.yes)}:${render(e.no)})"
}
fun fuzzYamlJson(value: Yaml): Json {
    if (value.isNull()) return jsonNull()
    if (value.isString()) return jsonString(value.asString())
    if (value.isBoolean()) return jsonBoolean(value.asBoolean())
    if (value.isNumber()) {
        if (value.intAvailable()) return jsonInteger(value.asInt())
        return jsonString("FLOAT:" + value.asNumber().text)
    }
    val values: array<Json> = []
    for (item in value.values) values.append(fuzzYamlJson(item))
    if (value.isSequence()) return jsonArray(values)
    val keys: array<string> = []
    for (key in value.keys) keys.append(key.asString())
    jsonObject(keys, values)
}

fun safeSlice(s: string, a: int, b: int): string {
    val lo = (a & 2147483647) % (s.size + 1)
    val hi = lo + ((b & 2147483647) % (s.size - lo + 1))
    s.slice(lo, hi)
}

fun drainGen(values: Generator<int>): int {
    var sum = 0

    sum
}

    sum
